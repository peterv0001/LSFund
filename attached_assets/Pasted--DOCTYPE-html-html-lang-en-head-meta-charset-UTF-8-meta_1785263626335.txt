function draw(){
  ctx.save();
  ctx.clearRect(0,0,canvas.width,canvas.height);
  ctx.scale(dpr,dpr);
  ctx.translate(transform.x,transform.y);
  ctx.scale(transform.k,transform.k);
  const k=transform.k;

  const focus = focusSet;
  // links
  for(const l of links){
    if(!linkVisible(l))continue;
    const dimmed = focus && !(focus.has(l.source.id)&&focus.has(l.target.id));
    ctx.beginPath();
    ctx.moveTo(l.source.x,l.source.y); ctx.lineTo(l.target.x,l.target.y);
    if(dimmed){ ctx.strokeStyle='rgba(40,69,92,.12)'; ctx.lineWidth=.7/k; }
    else if(focus){ ctx.strokeStyle='rgba(95,212,196,.55)'; ctx.lineWidth=1.4/k; }
    else{ ctx.strokeStyle='rgba(70,110,140,.28)'; ctx.lineWidth=.9/k; }
    ctx.stroke();
  }
  // pulses (loop mode)
  if(['engine','example','routing'].includes(mode) && !reduceMotion){
    for(const l of loopEdges()){
      const t=(pulseT + (hash(l.source.id+l.target.id)%1000)/1000)%1;
      const x=l.source.x+(l.target.x-l.source.x)*t, y=l.source.y+(l.target.y-l.source.y)*t;
      ctx.beginPath(); ctx.arc(x,y,2.6/Math.sqrt(k),0,7);
      ctx.fillStyle='rgba(126,224,163,.9)'; ctx.fill();
    }
  }
  // nodes
  for(const n of nodes){
    if(!visible(n))continue;
    const meta=TYPE[n.type]||{c:'#888'};
    const dimmed=focus&&!focus.has(n.id);
    const isSel=selected&&selected.id===n.id, isHov=hovered&&hovered.id===n.id;
    ctx.beginPath(); ctx.arc(n.x,n.y,n.r,0,7);
    ctx.fillStyle=dimmed?'rgba(30,48,64,.5)':meta.c;
    ctx.globalAlpha=dimmed?.5:1;
    ctx.fill();
    if(!dimmed){
      ctx.lineWidth=(isSel?2.4:1.1)/Math.sqrt(k);
      if(n.confidence==='high'){ ctx.setLineDash([]); }
      else{ ctx.setLineDash([3/k,3/k]); }
      ctx.strokeStyle=isSel?'#FFFFFF':(isHov?'rgba(255,255,255,.8)':'rgba(10,22,34,.7)');
      ctx.stroke(); ctx.setLineDash([]);
    }
    ctx.globalAlpha=1;
  }
  // labels
  ctx.textAlign='center'; ctx.textBaseline='top';
  for(const n of nodes){
    if(!visible(n))continue;
    const dimmed=focus&&!focus.has(n.id);
    if(dimmed)continue;
    const show = (selected&&selected.id===n.id)||(hovered&&hovered.id===n.id)||
                 n.deg>= (mode==='full'||mode==='overview'?11:3) || k>1.6 || !!focus;
    if(!show)continue;
    const fs=Math.max(9.5/k, 3);

  }

}

function pick(mx,my){
  const [gx,gy]=transform.invert([mx,my]);
  let best=null,bd=1e9;
  for(const n of nodes){ if(!visible(n))continue;
    const d=Math.hypot(n.x-gx,n.y-gy);
    if(d<n.r+6/transform.k && d<bd){bd=d;best=n;} }
  return best;
}

function renderInspector(n){
  const meta=TYPE[n.type]||{c:'#888'};
  const t=document.getElementById('insp-type');
  t.textContent=n.type; t.style.color=meta.c; t.style.background=meta.c+'22'; t.style.border=`1px solid ${meta.c}55`;
  document.getElementById('insp-title').textContent=n.label;
  document.getElementById('insp-id').textContent=n.id+'  ·  '+n.deg+' connections';
  const skip=new Set(['id','label','type','x','y','vx','vy','fx','fy','index','deg','r']);
  let html='';
  const props=Object.entries(n).filter(([k,v])=>!skip.has(k)&&v!==''&&v!=null);
  if(props.length){ html+='<div class="sect">Properties</div>';
    for(const[k,v]of props) html+=`<div class="prop"><b>${esc(k)}</b><span>${esc(v)}</span></div>`; }
  const out=links.filter(l=>l.source.id===n.id), inn=links.filter(l=>l.target.id===n.id);
  if(out.length){ html+=`<div class="sect">Outgoing · ${out.length}</div>`;
    for(const l of out){ const o=l.target,m=TYPE[o.type]||{c:'#888'};
      html+=`<div class="edge" data-nav="${esc(o.id)}"><span class="rel">${esc(l.relationship)}</span><span class="dir">→</span><span class="dot" style="background:${m.c}"></span><span class="lbl">${esc(o.label)}</span></div>`; } }
  if(inn.length){ html+=`<div class="sect">Incoming · ${inn.length}</div>`;

      html+=`<div class="edge" data-nav="${esc(o.id)}"><span class="rel">${esc(l.relationship)}</span><span class="dir">←</span><span class="dot" style="background:${m.c}"></span><span class="lbl">${esc(o.label)}</span></div>`; } }
  const body=document.getElementById('insp-body');
  body.innerHTML=html;
